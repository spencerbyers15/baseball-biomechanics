# Crontab entries for the FieldVision autopilot on Nellie (crontab -e).
# The keepalive relaunches the tmux session if it ever dies; the script
# itself is a no-op while the session is healthy or the NAS is unmounted.
# `bash` prefix is required: the NAS is mounted noexec.
#
# cron runs with a minimal env (no guaranteed $USER — see nellie_autopilot.sh),
# so resolve the account with `id -un` rather than trusting $USER here too.

*/10 * * * * bash /media/scratch/$(id -un)/github/baseball-biomechanics/fieldvision/scripts/nellie_autopilot.sh
@reboot sleep 90 && bash /media/scratch/$(id -un)/github/baseball-biomechanics/fieldvision/scripts/nellie_autopilot.sh
